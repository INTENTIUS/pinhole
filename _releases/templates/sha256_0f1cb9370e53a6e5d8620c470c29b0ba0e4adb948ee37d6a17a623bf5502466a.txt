name: ci

on:
  pull_request: {}
  push:
    branches:
      - main

permissions:
  contents: read

concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: '${{ github.ref != ''refs/heads/main'' }}'

jobs:
  check:
    name: check
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262
      - uses: actions/setup-node@49933ea5288caeca8642d1e84afbd3f7d6820020
        with:
          node-version: '22'
          cache: npm
      - run: npm ci
      - name: typecheck
        run: npm run tsc
      - name: test
        run: npm test
      - name: build
        run: npm run build
      - name: root entry bundles for a browser
        run: npm run check:browser
      - name: generated files match their sources
        run: npm run generated-check
